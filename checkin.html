function doPost(e) { try { const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheets()[0]; const data = JSON.parse(e.postData.contents); const guests = data.guests || []; const folderName = 'チェックイン_パスポート写真'; const folders = DriveApp.getFoldersByName(folderName); const folder = folders.hasNext() ? folders.next() : DriveApp.createFolder(folderName); const row = [new Date()]; for (let i = 0; i < 6; i++) { const g = guests[i] || {}; let photoUrl = ''; if (g.photo) { const parts = g.photo.split(','); const blob = Utilities.newBlob( Utilities.base64Decode(parts[1]), 'image/jpeg', 'passport_' + (g.name || 'guest') + '_' + Date.now() + '_' + i + '.jpg' ); const file = folder.createFile(blob); photoUrl = file.getUrl(); } row.push(g.name || '', g.passport || '', photoUrl); } sheet.appendRow(row); return ContentService.createTextOutput(JSON.stringify({ok: true})) .setMimeType(ContentService.MimeType.JSON); } catch (err) { return ContentService.createTextOutput(JSON.stringify({ok: false, error: err.toString()})) .setMimeType(ContentService.MimeType.JSON); } }
